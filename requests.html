async function submitRequest(song) { showToast(`Processing: ${song.artist} - ${song.title}...`, "info"); try { const response = await fetch(PROXY_URL, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ artist: song.artist, title: song.title // No necesitas enviar email, el Worker lo pone }) }); const result = await response.json(); if (result.success) { showToast(`✓ "${song.title}" by ${song.artist} added to queue.`, "success"); } else { showToast(`✗ ${result.message || `Could not add "${song.title}"`}`, "error"); } } catch (error) { showToast(`Connection error: ${error.message}`, "error"); } }
